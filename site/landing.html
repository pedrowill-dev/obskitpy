No seu código

Três linhas e o logging de sempre

Spans com um decorator, métricas de negócio com a API padrão do OpenTelemetry e erros com contexto automático.

import logging, obskitpy

log = logging.getLogger(__name__)
shutdown = obskitpy.setup("meu-servico")   # logs+traces+métricas

pedidos = obskitpy.meter("pedidos").create_counter("pedidos.processados")

@obskitpy.traced()                          # um span por chamada
def processar(item):
    ...

try:
    processar(x)
    pedidos.add(1, {"status": "success"})
except Exception:
    log.exception("falhou")                 # + error.type/origin
    pedidos.add(1, {"status": "failure"})

shutdown()                                  # descarrega os pendentes
